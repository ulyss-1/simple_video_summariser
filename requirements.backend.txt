# Runtime Python dependencies for the backend services (api, planner,
# analyzer, migrate, and the transcriber base image).
#
# Runtime deps only, per AGENTS.md -> Rules:
# - speech-to-text-only deps belong in requirements.whisper.txt
# - dev tooling (pytest, hypothesis, ruff, mypy) belongs in the `dev` extra
#   of pyproject.toml
# - frontend deps belong in web/package.json
#
# Each dependency is added by the first task that needs it. Ask before
# adding one.

# YouTube metadata, subtitles and audio. Pinned exactly; a scheduled update
# check keeps it fresh (architecture.md 16.11).
yt-dlp==2026.10.21

# yt-dlp's YouTube challenge solver scripts, shipped as a package so nothing is
# fetched at runtime (#134). Must equal the yt-dlp-ejs==X that the pinned yt-dlp
# requires in its `default` extra (0.8.0 for yt-dlp 2026.10.21); it moves with
# yt-dlp and with the Deno pin in ops/Dockerfile.backend.
yt-dlp-ejs==0.8.0

# Typed, validated settings from environment variables (architecture.md 10,
# common/config.py). Pydantic itself comes in transitively.
pydantic-settings==2.16.1

# Structured JSON logs to stdout, stdlib records included (architecture.md 2,
# 12; common/logging.py).
structlog==26.1.0

# Migrations, hand-written SQL only, target_metadata=None (architecture.md 6,
# 11.4; task #7). Pulls in SQLAlchemy transitively, used only as Alembic's
# connection/engine layer, not as an ORM.
alembic==1.20.1

# Postgres driver. The compose DSN uses the postgresql:// scheme, which
# SQLAlchemy maps to the (uninstalled) psycopg2 by default, so
# migrations/env.py rewrites it to postgresql+psycopg:// (task #7).
psycopg[binary]==3.3.6

# Connection pool for the API only (task #95): sync ConnectionPool, pinned to
# the release that resolves with psycopg 3.3.6.
psycopg-pool==3.3.3

# HTTP API (architecture.md 8.3, 16.10; task #39). Plain uvicorn, without the
# [standard] extras: the API serves a single operator, so uvloop/httptools
# buy nothing measurable.
fastapi==0.162.0
uvicorn==0.54.0
